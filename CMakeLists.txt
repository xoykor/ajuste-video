cmake_minimum_required(VERSION 3.20)
project(ajuste-video VERSION 0.1.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(QT_MAJOR_VERSION 6)
include(GNUInstallDirs)

option(BUILD_KWIN_EFFECT "Build the KWin compositor effect" ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus)

if(BUILD_KWIN_EFFECT)
    find_package(KF6Config REQUIRED)
    find_package(ECM REQUIRED NO_MODULE)
    set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} ${ECM_MODULE_PATH})
    include(KDEInstallDirs)
    include(KDECMakeSettings)
    find_package(KWin REQUIRED)

    add_library(ajustevideo MODULE effect.cpp effect.h effect.qrc effectplugin.cpp metadata.json)
    target_link_libraries(ajustevideo PRIVATE Qt6::Core Qt6::Gui KF6::ConfigCore KWin::kwin)
    target_include_directories(ajustevideo PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    configure_file(metadata.json ${CMAKE_CURRENT_BINARY_DIR}/metadata.json COPYONLY)
    install(TARGETS ajustevideo DESTINATION ${KDE_INSTALL_PLUGINDIR}/kwin/effects/plugins)
else()
    include(GNUInstallDirs)
    set(KDE_INSTALL_BINDIR ${CMAKE_INSTALL_BINDIR})
    set(KDE_INSTALL_APPDIR ${CMAKE_INSTALL_DATADIR}/applications)
    set(KDE_INSTALL_ICONDIR ${CMAKE_INSTALL_DATADIR}/icons)
endif()

add_executable(ajuste-video main.cpp)
target_link_libraries(ajuste-video PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus)
install(TARGETS ajuste-video DESTINATION ${KDE_INSTALL_BINDIR})
install(FILES ajuste-video.desktop DESTINATION ${KDE_INSTALL_APPDIR})
install(FILES ajuste-video.svg DESTINATION ${KDE_INSTALL_ICONDIR}/hicolor/scalable/apps)
install(DIRECTORY backends/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ajuste-video/backends)
install(FILES CMakeLists.txt effect.cpp effect.h effect.qrc effectplugin.cpp metadata.json ajuste-video.desktop ajuste-video.svg
        DESTINATION ${CMAKE_INSTALL_DATADIR}/ajuste-video/source)
install(DIRECTORY shaders/ DESTINATION ${CMAKE_INSTALL_DATADIR}/ajuste-video/source/shaders)
